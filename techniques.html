🔒 1. 臨界區保護（Mutex / Spinlock / Atomic）▼
核心任務：多個執行緒對同一個變數（如 counter）進行加減。如果沒有保護，會發生 Lost Update（結果偏小）。
成大考點：PDF 演示了 no lock / lock short / lock long 的差別。鎖要抓得越短越好，只包住真正存取的變數。
參考資料： TLPI thread_lock_speed.c
骨架 A：標準 pthread_mutex（適合鎖內有耗時操作） 
#include <stdio.h>
#include <pthread.h>

int counter = 0;
pthread_mutex_t lock = PTHREAD_MUTEX_INITIALIZER;

void* worker(void* arg) {
    for (int i = 0; i < 20000; i++) {
        pthread_mutex_lock(&lock);
        counter++; // 臨界區：只包這行
        pthread_mutex_unlock(&lock);
    }
    return NULL;
}

int main() {
    pthread_t t1, t2;
    pthread_create(&t1, NULL, worker, NULL);
    pthread_create(&t2, NULL, worker, NULL);
    pthread_join(t1, NULL);
    pthread_join(t2, NULL);
    printf("counter = %d\n", counter);
    return 0;
}
骨架 B：pthread_spinlock（適合極短臨界區） 
#include <stdio.h>
#include <pthread.h>

int counter = 0;
pthread_spinlock_t spinlock;

void* worker(void* arg) {
    for (int i = 0; i < 20000; i++) {
        pthread_spin_lock(&spinlock);
        counter++;
        pthread_spin_unlock(&spinlock);
    }
    return NULL;
}

int main() {
    pthread_t t1, t2;
    pthread_spin_init(&spinlock, 0); // 0 = 執行緒間共享
    pthread_create(&t1, NULL, worker, NULL);
    pthread_create(&t2, NULL, worker, NULL);
    pthread_join(t1, NULL);
    pthread_join(t2, NULL);
    pthread_spin_destroy(&spinlock);
    printf("counter = %d\n", counter);
    return 0;
}
骨架 C：C11 原子操作（無鎖，最快，但要注意 C11 支援） 
#include <stdio.h>
#include <pthread.h>
#include <stdatomic.h>

atomic_int counter = 0;

void* worker(void* arg) {
    for (int i = 0; i < 20000; i++) {
        atomic_fetch_add_explicit(&counter, 1, memory_order_relaxed);
    }
    return NULL;
}

int main() {
    pthread_t t1, t2;
    pthread_create(&t1, NULL, worker, NULL);
    pthread_create(&t2, NULL, worker, NULL);
    pthread_join(t1, NULL);
    pthread_join(t2, NULL);
    printf("counter = %d\n", atomic_load(&counter));
    return 0;
}
🚨 致命錯誤：絕對不要把 printf、sleep、malloc 或大量運算放進鎖裡面！這會導致其他執行緒全部卡死等待，效能比單執行緒還差（PDF 實測：lock long 耗時 155.6ms）。
📦 2. 生產者-消費者模型（Queue + Condvar）▼
核心任務：一個（或多個）執行緒產生資料，另一個（或多個）執行緒消耗資料。
必考點：為什麼 pthread_cond_wait 必須放在 while 迴圈裡？因為有 Spurious Wakeup（虛假喚醒）。
參考資料： CThreads repo
純 C 環形緩衝區 + Mutex + Condvar（完整可編譯） 
#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>

#define QUEUE_SIZE 10
int queue[QUEUE_SIZE];
int head = 0, tail = 0, count = 0;

pthread_mutex_t mutex = PTHREAD_MUTEX_INITIALIZER;
pthread_cond_t not_empty = PTHREAD_COND_INITIALIZER;
pthread_cond_t not_full = PTHREAD_COND_INITIALIZER;

void* producer(void* arg) {
    for (int i = 0; i < 100; i++) {
        pthread_mutex_lock(&mutex);
        while (count == QUEUE_SIZE) {
            pthread_cond_wait(&not_full, &mutex);
        }
        queue[tail] = i;
        tail = (tail + 1) % QUEUE_SIZE;
        count++;
        pthread_cond_signal(&not_empty);
        pthread_mutex_unlock(&mutex);
    }
    return NULL;
}

void* consumer(void* arg) {
    for (int i = 0; i < 100; i++) {
        pthread_mutex_lock(&mutex);
        while (count == 0) {
            pthread_cond_wait(&not_empty, &mutex);
        }
        int val = queue[head];
        head = (head + 1) % QUEUE_SIZE;
        count--;
        pthread_cond_signal(&not_full);
        pthread_mutex_unlock(&mutex);
        printf("Consumed: %d\n", val);
    }
    return NULL;
}

int main() {
    pthread_t p, c;
    pthread_create(&p, NULL, producer, NULL);
    pthread_create(&c, NULL, consumer, NULL);
    pthread_join(p, NULL);
    pthread_join(c, NULL);
    return 0;
}
💡 結束條件設計：如果生產者結束後消費者還在等，會死鎖。常見做法是生產者最後放入「毒藥（Poison Pill）」（例如 -1），或者用一個 done 旗標，並搭配 pthread_cond_broadcast 喚醒所有執行緒。
🚚 3. 資料搬運引擎（DMA 四拍節奏）▼
成大 PDF 考點：描述（configure）→ 啟動（start）→ 等待（wait）→ 取用結果。
致命陷阱：count 的單位是「幾筆傳輸」，不是「幾個位元組」。每筆大小只能是 1、2 或 4 bytes。
參考資料： Pico SDK DMA 中文範例
骨架 A：概念模擬（用 C 結構體示範） 
#include <stdio.h>
#include <stdint.h>
#include <string.h>

typedef struct {
    void* src;
    void* dst;
    uint32_t count;       // 幾筆傳輸（不是 bytes！）
    uint32_t element_size; // 每筆大小：1, 2, 或 4 bytes
    uint32_t src_inc;      // 來源位址是否遞增 (1=是, 0=否)
    uint32_t dst_inc;      // 目標位址是否遞增 (1=是, 0=否)
} dma_transfer_t;

void simulate_dma(dma_transfer_t* t) {
    char* s = (char*)t->src;
    char* d = (char*)t->dst;
    for (uint32_t i = 0; i < t->count; i++) {
        memcpy(d, s, t->element_size);
        if (t->src_inc) s += t->element_size;
        if (t->dst_inc) d += t->element_size;
    }
}

int main() {
    uint32_t src[4] = {0x11111111, 0x22222222, 0x33333333, 0x44444444};
    uint32_t dst[4] = {0};
    dma_transfer_t t = { src, dst, 4, 4, 1, 1 };
    simulate_dma(&t);
    return 0;
}
骨架 B：Pico SDK DMA 實戰 API（考場填空用） 
#include "pico/stdlib.h"
#include "hardware/dma.h"

// 1. 獲取一個空閒通道，如果沒有則 panic
int chan = dma_claim_unused_channel(true);

// 2. 獲取該通道的默認配置
dma_channel_config c = dma_channel_get_default_config(chan);

// 3. 設定傳輸資料大小（DMA_SIZE_8 / _16 / _32）
channel_config_set_transfer_data_size(&c, DMA_SIZE_8);

// 4. 設定讀/寫位址是否自增（true = 每次搬完指標往前）
channel_config_set_read_increment(&c, true);
channel_config_set_write_increment(&c, true);

// 5. 寫入通道配置並啟動
// 參數：(通道, 配置, 目標, 來源, 傳輸次數, 是否立即啟動)
dma_channel_configure(
    chan,
    &c,
    dst,
    src,
    count_of(src), // 注意：是幾筆，不是幾個 bytes
    true
);

// 6. 等待 DMA 完成（阻塞式）
dma_channel_wait_for_finish_blocking(chan);
🔢 4. 位元運算（Bitwise）▼
成大 PDF 考點：設定、清除、取出、遮罩。注意移位時加上 u 避免有號整數溢位。
必背位元操作巨集 
#include <stdio.h>
#include <stdint.h>

#define SET_BIT(x, n)      ((x) |= (1u << (n)))
#define CLEAR_BIT(x, n)    ((x) &= ~(1u << (n)))
#define TOGGLE_BIT(x, n)   ((x) ^= (1u << (n)))
#define GET_BIT(x, n)      (((x) >> (n)) & 1u)
#define GET_BITS(x, hi, lo) (((x) >> (lo)) & ((1u << ((hi) - (lo) + 1)) - 1))

int main() {
    uint32_t x = 0xA5; // 1010 0101
    SET_BIT(x, 1);
    printf("GET_BITS = 0x%X\n", GET_BITS(0xA5, 3, 0));
    return 0;
}
⚙️ 5. PIO 狀態機（Pico SDK）▼
成大 PDF 考點：CPU 透過 TX FIFO 送資料進 OSR，程式以 out 取出；以 in 放入 ISR，再以 push 送回 CPU。
參考資料： 官方 hello_pio 範例
PIO 載入與啟動最小模板 
#include "pico/stdlib.h"
#include "hardware/pio.h"
#include "hello.pio.h" // 編譯時自動生成的標頭檔

int main() {
    PIO pio;
    uint sm;
    uint offset;

    // 1. 獲取空閒的 PIO 與狀態機，並載入程式
    bool success = pio_claim_free_sm_and_add_program_for_gpio_range(
        &hello_program, &pio, &sm, &offset, PIN, 1, true
    );
    hard_assert(success);

    // 2. 初始化該狀態機（通常會配置 GPIO 方向等）
    hello_program_init(pio, sm, offset, PIN);

    // 3. 將資料推入 TX FIFO（送到 OSR）
    pio_sm_put_blocking(pio, sm, 0x12345678);

    // 4. 從 RX FIFO 取出資料（從 ISR 來）
    // uint32_t data = pio_sm_get_blocking(pio, sm);

    // 5. 釋放資源
    pio_remove_program_and_unclaim_sm(&hello_program, pio, sm, offset);
    return 0;
}
🤖 6. ROS 2 節點骨架與排查 SOP▼
成大考場重點：ROS 2 Jazzy 環境。不會叫你從零寫整個機器人，但會考「節點通訊」與「資料沒來怎麼查」。
參考資料： 官方 Python Pub/Sub 教程、Docker 執行 Talker/Listener
ROS 2 Python 節點骨架（Publisher / Subscriber） 
import rclpy
from rclpy.node import Node
from std_msgs.msg import Int32

class MyNode(Node):
    def __init__(self):
        super().__init__('my_node')
        self.pub = self.create_publisher(Int32, '/number', 10)
        self.sub = self.create_subscription(Int32, '/double', self.callback, 10)
        self.timer = self.create_timer(1.0, self.timer_callback)
        self.i = 0

    def timer_callback(self):
        msg = Int32()
        msg.data = self.i
        self.pub.publish(msg)
        self.i += 1

    def callback(self, msg):
        self.get_logger().info(f'Received: {msg.data}')

def main(args=None):
    rclpy.init(args=args)
    node = MyNode()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        node.destroy_node()
        rclpy.shutdown()

if __name__ == '__main__':
    main()
ROS 2 故障排查決策樹（成大 PDF 第五點實戰） 
【ROS 2 資料沒來時的排查順序】
請依照以下順序，在終端機輸入指令確認：

1. 確認 Node 存在
   $ ros2 node list
   （如果沒看到，代表程式根本沒啟動或崩潰了）

2. 確認 Topic 存在
   $ ros2 topic list
   （如果沒看到，代表名稱拼錯，或 Node 沒建立發佈/訂閱）

3. 確認兩端是否連到相同的 Topic
   $ ros2 node info /你的節點名稱
   （檢查 Publisher 與 Subscriber 的 Topic 名稱是否完全一致）

4. 確認 Publisher/Subscriber 數量是否合理
   $ ros2 topic info /topic名稱
   （例如應該有 1 個發佈、1 個訂閱，如果是 0 就代表沒接上）

5. 確認實際是否有資料傳輸
   $ ros2 topic echo /topic名稱
   （看有沒有資料，如果沒有，代表上游根本沒發）

6. 確認頻率
   $ ros2 topic hz /topic名稱
   （確認是否有卡頓、掉幀）
💡 實用小技巧：如果不想寫 Publisher 程式，可以直接用終端機發送測試訊息： ros2 topic pub --once /number std_msgs/msg/Int32 "{data: 21}"
